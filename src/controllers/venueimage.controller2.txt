import { Request, Response } from "express";
import { uploadVenueImage } from "../utils/supabase";
import VenueImageSvc from "../services/venueImage.service";

export default class VenueImageCtrl {
    static async uploadVenueImages(req: Request, res: Response) {
        try {
            const userId = (req as any).user?.userId;
            const { id } = req.params; // venueId
            const files = req.files as Express.Multer.File[];

            console.log(`📸 [VenueImageCtrl] Upload Attempt - Venue: ${id}, User: ${userId}, Files: ${files?.length || 0}`);

            if (!userId) {
                console.error("❌ [VenueImageCtrl] Upload failed: User not authenticated");
                return res.status(401).json({ success: false, message: "User not authenticated" });
            }

            if (!files || files.length === 0) {
                console.error("❌ [VenueImageCtrl] Upload failed: No images provided in request body");
                return res.status(400).json({ success: false, message: "No images uploaded" });
            }

            const images = [];
            for (let i = 0; i < files.length; i++) {
                const file = files[i];
                console.log(`📤 [VenueImageCtrl] Processing file ${i + 1}/${files.length}: ${file.originalname} (${file.size} bytes)`);

                try {
                    // Upload to Supabase
                    console.log(`   - Uploading to Supabase bucket...`);
                    const url = await uploadVenueImage(file, id);
                    console.log(`   - Supabase success: ${url}`);

                    const isThumbnail = i === 0;
                    console.log(`   - Saving to database (VenueImageSvc.addImage)...`);
                    const image = await VenueImageSvc.addImage(id, userId, url, isThumbnail, file.originalname, i);
                    console.log(`   - Database success: Image ID ${image.id}`);
                    images.push(image);
                } catch (innerError: any) {
                    console.error(`   - ❌ Failed during processing of file ${file.originalname}:`, innerError.message);
                    throw innerError;
                }
            }

            console.log("✅ [VenueImageCtrl] All images uploaded and saved successfully");
            return res.status(201).json({ success: true, data: images });
        } catch (error: any) {
            console.error("❌ [VenueImageCtrl] Final Catch Error:", error.message);
            return res.status(400).json({
                success: false,
                message: error.message,
                error: error.message,
                stack: error.stack
            });
        }
    }

    static async updateVenueImage(req: Request, res: Response) {
        try {
            const userId = (req as any).user?.userId;
            const { imageId } = req.params;
            const data = req.body;

            if (!userId) {
                return res.status(401).json({ success: false, message: "User not authenticated" });
            }

            const image = await VenueImageSvc.updateImage(userId, imageId, data);
            return res.status(200).json({ success: true, data: image });
        } catch (error: any) {
            return res.status(400).json({ success: false, message: error.message });
        }
    }

    static async deleteVenueImage(req: Request, res: Response) {
        try {
            const userId = (req as any).user?.userId;
            const { imageId } = req.params;

            if (!userId) {
                return res.status(401).json({ success: false, message: "User not authenticated" });
            }

            await VenueImageSvc.deleteImage(userId, imageId);
            return res.status(200).json({ success: true, message: "Image deleted successfully" });
        } catch (error: any) {
            return res.status(400).json({ success: false, message: error.message });
        }
    }
}
